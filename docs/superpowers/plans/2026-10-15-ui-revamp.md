// public/app-shell.js
export const WORKSPACE_VIEWS = ['capture', 'editor', 'reading'];

export function resolveView(requestedView, { hasProject = false } = {}) {
  if (!hasProject) {
    return 'library';
  }
  return WORKSPACE_VIEWS.includes(requestedView) ? requestedView : 'library';
}

export function buildShellState({ view, project = null, busy = false, system = null } = {}) {
  const resolvedView = resolveView(view, { hasProject: Boolean(project) });
  const pageCount = project?.pages?.length || 0;

  return {
    view: resolvedView,
    topbar: resolvedView === 'library' ? 'library' : 'workspace',
    exportDisabled: !project || pageCount === 0 || Boolean(busy),
    bookStatus: project ? `${pageCount} ${pageCount === 1 ? 'página' : 'páginas'}` : 'Sin libro abierto',
    settingsAlert: Boolean(system?.update?.available)
  };
}
```

// public/ui-popover.js
const openPopovers = new Set();

export function setupPopover(trigger, panel, { onOpen } = {}) {
  trigger.setAttribute('aria-haspopup', 'true');
  trigger.setAttribute('aria-expanded', 'false');
  if (panel.id) {
    trigger.setAttribute('aria-controls', panel.id);
  }
  panel.hidden = true;

  const controller = {
    isOpen: () => !panel.hidden,
    open() {
      closeAllPopovers();
      panel.hidden = false;
      trigger.setAttribute('aria-expanded', 'true');
      openPopovers.add(controller);
      onOpen?.();
      const firstFocusable = panel.querySelector(
        'input:not([disabled]), select:not([disabled]), button:not([disabled]), [href]'
      );
      firstFocusable?.focus();
    },
    close({ restoreFocus = true } = {}) {
      if (panel.hidden) {
        return;
      }
      panel.hidden = true;
      trigger.setAttribute('aria-expanded', 'false');
      openPopovers.delete(controller);
      if (restoreFocus) {
        trigger.focus();
      }
    },
    contains: (node) => trigger.contains(node) || panel.contains(node)
  };

  trigger.addEventListener('click', (event) => {
    event.stopPropagation();
    controller.isOpen() ? controller.close() : controller.open();
  });

  panel.addEventListener('keydown', (event) => {
    if (event.key === 'Escape') {
      // Evita que un <dialog> padre se cierre con el mismo Esc.
      event.preventDefault();
      event.stopPropagation();
      controller.close();
    }
  });

  return controller;
}

export function closeAllPopovers() {
  for (const controller of [...openPopovers]) {
    controller.close({ restoreFocus: false });
  }
}

document.addEventListener('click', (event) => {
  for (const controller of [...openPopovers]) {
    if (!controller.contains(event.target)) {
      controller.close({ restoreFocus: false });
    }
  }
});

export function openDrawer(dialogId, { section } = {}) {
  const dialog = document.getElementById(dialogId);
  if (!dialog) {
    return null;
  }
  closeAllPopovers();
  if (!dialog.open) {
    dialog.showModal();
  }
  if (section) {
    const details = document.getElementById(section);
    if (details instanceof HTMLDetailsElement) {
      details.open = true;
    }
    details?.scrollIntoView({ block: 'start' });
    return details;
  }
  return dialog;
}

export function setupDrawers(root = document) {
  for (const trigger of root.querySelectorAll('[data-open-drawer]')) {
    trigger.addEventListener('click', () => openDrawer(trigger.dataset.openDrawer));
  }
  for (const closer of root.querySelectorAll('[data-close-drawer]')) {
    closer.addEventListener('click', () => closer.closest('dialog')?.close());
  }
  for (const dialog of root.querySelectorAll('dialog.drawer')) {
    // Clic en el backdrop: el target es el propio <dialog>.
    dialog.addEventListener('click', (event) => {
      if (event.target === dialog) {
        dialog.close();
      }
    });
  }
}
```

// public/page-range-selection.js
export function extendSelection(orderedIds, selectedIds, anchorId, targetId, checked) {
  const next = new Set(selectedIds);
  const targetIndex = orderedIds.indexOf(targetId);
  const anchorIndex = anchorId ? orderedIds.indexOf(anchorId) : -1;
  const span =
    anchorIndex >= 0 && targetIndex >= 0
      ? orderedIds.slice(Math.min(anchorIndex, targetIndex), Math.max(anchorIndex, targetIndex) + 1)
      : [targetId];

  for (const id of span) {
    if (checked) {
      next.add(id);
    } else {
      next.delete(id);
    }
  }
  return next;
}

export function selectionBounds(pages, selectedIds) {
  const numbers = (pages || [])
    .filter((page) => selectedIds.has(page.id))
    .map((page) => Number(page.number))
    .filter(Number.isFinite);

  if (!numbers.length) {
    return null;
  }
  return { start: Math.min(...numbers), end: Math.max(...numbers) };
}

export function adjacentPageId(visiblePages, currentId, direction) {
  const list = visiblePages || [];
  if (!list.length) {
    return null;
  }
  const index = list.findIndex((page) => page.id === currentId);
  if (index < 0) {
    return list[0].id;
  }
  return list[index + direction]?.id || null;
}

- [ ] **Step 3: Write minimal implementation**

```js
// public/capture-session.js
export function startCaptureSession(pages = []) {
  return { baselineIds: (pages || []).map((page) => page.id) };
}

export function summarizeCaptureSession(session, pages = []) {
  const empty = { count: 0, newPageIds: [], firstPageId: null, contiguous: false, range: null };
  if (!session) {
    return empty;
  }

  const baseline = new Set(session.baselineIds || []);
  const newPages = (pages || []).filter((page) => !baseline.has(page.id));
  if (!newPages.length) {
    return empty;
  }

  const numbers = newPages.map((page) => Number(page.number)).sort((a, b) => a - b);
  const contiguous = numbers.every((value, index) => index === 0 || value === numbers[index - 1] + 1);

  return {
    count: newPages.length,
    newPageIds: newPages.map((page) => page.id),
    firstPageId: newPages[0].id,
    contiguous,
    range: contiguous ? { start: numbers[0], end: numbers[numbers.length - 1] } : null
  };
}
```

// public/library-dashboard.js (añadir al final)
function plural(count, singular, pluralForm) {
  return `${count} ${count === 1 ? singular : pluralForm}`;
}

export function formatLibrarySummaryLine(summary = {}) {
  return [
    plural(Number(summary.bookCount || 0), 'libro', 'libros'),
    plural(Number(summary.pageCount || 0), 'página', 'páginas'),
    `${Number(summary.averageReviewedPercent || 0)} % revisado`,
    plural(Number(summary.pendingProblemCount || 0), 'problema', 'problemas')
  ].join(' · ');
}

export function libraryCardCover(project = {}) {
  const mode = project.cover?.mode;
  if (!project.id || !mode || mode === 'none') {
    return { url: null };
  }
  const version = project.cover.updatedAt || project.updatedAt || '';
  return {
    url: `/api/projects/${encodeURIComponent(project.id)}/cover/image?${encodeURIComponent(version)}`
  };
}
